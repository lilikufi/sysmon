function showToast(message, duration = 2000) {
    const existingToast = document.querySelector('.copy-toast');
    if (existingToast) {
        existingToast.remove();
    }

    const toast = document.createElement('div');
    toast.className = 'copy-toast';
    toast.textContent = message;
    document.body.appendChild(toast);

    setTimeout(() => toast.classList.add('show'), 10);

    setTimeout(() => {
        toast.classList.remove('show');
        setTimeout(() => toast.remove(), 300);
    }, duration);
}

// Копировать в буфер обмена
async function copyToClipboard(text) {
    try {
        await navigator.clipboard.writeText(text);
        return true;
    } catch (err) {
        const textarea = document.createElement('textarea');
        textarea.value = text;
        textarea.style.position = 'fixed';
        textarea.style.opacity = '0';
        document.body.appendChild(textarea);
        textarea.select();
        try {
            document.execCommand('copy');
            document.body.removeChild(textarea);
            return true;
        } catch (e) {
            document.body.removeChild(textarea);
            return false;
        }
    }
}

// Открыть SSH через сервер (работает на Linux и Windows)
async function openSSHTerminal(ip, protocol = 'ssh') {
    showToast(`Открытие ${protocol.toUpperCase()} соединения к ${ip}...`);

    try {
        const response = await fetch('/api/open-terminal/', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRFToken': getCookie('csrftoken')
            },
            body: JSON.stringify({
                ip: ip,
                protocol: protocol
            })
        });

        const data = await response.json();

        if (data.success) {
            showToast(data.message || `${protocol.toUpperCase()} терминал открыт`);
        } else {
            showToast(data.message || 'Не удалось открыть терминал', 3000);
            offerAlternatives(ip, protocol);
        }
    } catch (error) {
        console.error('Error opening terminal:', error);
        showToast('Ошибка подключения к серверу', 2000);
        offerAlternatives(ip, protocol);
    }
}

// Предложить альтернативные способы подключения
function offerAlternatives(ip, protocol) {
    const command = protocol === 'ssh' ? `ssh root@${ip}` : `telnet ${ip}`;

    const modal = document.createElement('div');
    modal.className = 'terminal-help-modal';
    modal.innerHTML = `
        <div class="terminal-help-content">
            <h4>Подключение к ${ip}</h4>
            <p>Выполните команду в терминале:</p>
            <div class="terminal-command">
                <code>${command}</code>
                <button class="copy-cmd-btn" onclick="copyToClipboard('${command}').then(() => showToast('Команда скопирована'))">
                    📋 Копировать
                </button>
            </div>
            <div class="terminal-help-actions">
                <button onclick="this.closest('.terminal-help-modal').remove()">Закрыть</button>
            </div>
        </div>
    `;
    document.body.appendChild(modal);

    modal.addEventListener('click', (e) => {
        if (e.target === modal) modal.remove();
    });
}

// Инициализация SSH dropdown
function initSSHDropdown() {
    const dropdown = document.getElementById('sshDropdown');
    const button = document.getElementById('sshButton');

    if (!dropdown || !button) return;

    button.addEventListener('click', (e) => {
        e.preventDefault();
        e.stopPropagation();
        dropdown.classList.toggle('active');
    });

    document.addEventListener('click', (e) => {
        if (!dropdown.contains(e.target)) {
            dropdown.classList.remove('active');
        }
    });

    // SSH
    document.getElementById('sshOpenPutty')?.addEventListener('click', (e) => {
        e.preventDefault();
        if (currentHost) {
            openSSHTerminal(currentHost.data.ipaddr, 'ssh');
            dropdown.classList.remove('active');
        }
    });

    // Telnet
    document.getElementById('sshOpenTelnet')?.addEventListener('click', (e) => {
        e.preventDefault();
        if (currentHost) {
            openSSHTerminal(currentHost.data.ipaddr, 'telnet');
            dropdown.classList.remove('active');
        }
    });

    // Копировать SSH команду
    document.getElementById('sshCopyCommand')?.addEventListener('click', async (e) => {
        e.preventDefault();
        if (currentHost) {
            const command = `ssh root@${currentHost.data.ipaddr}`;
            const success = await copyToClipboard(command);
            showToast(success ? `Скопировано: ${command}` : 'Ошибка копирования');
            dropdown.classList.remove('active');
        }
    });

    // Копировать IP
    document.getElementById('sshCopyIp')?.addEventListener('click', async (e) => {
        e.preventDefault();
        if (currentHost) {
            const success = await copyToClipboard(currentHost.data.ipaddr);
            showToast(success ? `IP скопирован: ${currentHost.data.ipaddr}` : 'Ошибка копирования');
            dropdown.classList.remove('active');
        }
    });
}
