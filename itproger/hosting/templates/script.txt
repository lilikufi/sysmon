function getDeviceTypeConfig(type) {
    if (!type) return DEVICE_TYPE_CONFIG['unknown'];
    const normalizedType = type.toLowerCase().replace(/[-_\s]/g, '');
    for (const [key, config] of Object.entries(DEVICE_TYPE_CONFIG)) {
        if (normalizedType.includes(key.toLowerCase().replace(/[-_\s]/g, ''))) {
            return config;
        }
    }
    return DEVICE_TYPE_CONFIG['other'];
}

function buildGraphData(rawData) {
    console.log("=== ПОСТРОЕНИЕ ГРАФА ===");

    if (!rawData || !Array.isArray(rawData)) {
        console.error("Данные не являются массивом!");
        return { nodes: [], links: [] };
    }

    // Создаём узлы - КАЖДЫЙ элемент данных становится узлом
    const nodes = [];
    const nodeMap = new Map();

    rawData.forEach((item, index) => {
        if (!item.ipaddr) {
            console.warn(`Элемент ${index} без ipaddr:`, item);
            return;
        }

        const node = {
            id: item.ipaddr,
            data: item,
            x: Math.random() * 800 + 100,  // Случайная начальная позиция
            y: Math.random() * 600 + 100,
            fx: null,
            fy: null
        };

        nodes.push(node);
        nodeMap.set(item.ipaddr, node);
    });

    console.log("Создано узлов:", nodes.length);

    // Создаём связи
    const links = [];
    const addedLinks = new Set();

    rawData.forEach(item => {
        if (!item.ipaddr) return;

        // Связь parent -> current
        if (item.parent && nodeMap.has(item.parent)) {
            const linkKey = [item.parent, item.ipaddr].sort().join('|');
            if (!addedLinks.has(linkKey)) {
                links.push({
                    source: item.parent,
                    target: item.ipaddr
                });
                addedLinks.add(linkKey);
            }
        }

        // Связи current -> children
        if (item.children && Array.isArray(item.children)) {
            item.children.forEach(childIp => {
                if (typeof childIp === 'string' && nodeMap.has(childIp)) {
                    const linkKey = [item.ipaddr, childIp].sort().join('|');
                    if (!addedLinks.has(linkKey)) {
                        links.push({
                            source: item.ipaddr,
                            target: childIp
                        });
                        addedLinks.add(linkKey);
                    }
                }
            });
        }
    });

    console.log("Создано связей:", links.length);

    return { nodes, links };
}

const adjacency = new Map();
links.forEach(l => {
    const a = typeof l.source === 'object' ? l.source.id : l.source;
    const b = typeof l.target === 'object' ? l.target.id : l.target;
    adjacency.set(`${a}|${b}`, true);
    adjacency.set(`${b}|${a}`, true);
});

function isConnected(a, b) {
    const aId = a.id || (a.data && a.data.ipaddr);
    const bId = b.id || (b.data && b.data.ipaddr);
    return aId === bId || adjacency.has(`${aId}|${bId}`);
}

async function loadPositionsFromServer() {
    try {
        const r = await fetch(POSITIONS_URL, { credentials: 'same-origin' });
        if (!r.ok) return {};
        const positions = await r.json();
        console.log('Загружено позиций:', Object.keys(positions).length);
        return positions;
    } catch (e) {
        console.warn('Ошибка загрузки позиций:', e);
        return {};
    }
}

function savePositionToServer(ipaddr, x, y) {
    if (!Number.isFinite(x) || !Number.isFinite(y)) return Promise.resolve();

    return fetch(POSITIONS_URL, {
        method: 'POST',
        credentials: 'same-origin',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': CSRF
        },
        body: JSON.stringify({ ipaddr, x, y })
    })
    .then(response => {
        if (response.ok) {
            savedPositions[ipaddr] = { x, y };
        }
        return response;
    })
    .catch(e => console.warn('Ошибка сохранения позиции:', e));
}

function applyLoadedPositions(positions) {
    let count = 0;
    nodes.forEach(n => {
        const p = positions[n.id];
        if (p && Number.isFinite(p.x) && Number.isFinite(p.y)) {
            n.fx = p.x;
            n.fy = p.y;
            n.x = p.x;
            n.y = p.y;
            count++;
        }
    });
    console.log('Применено позиций:', count);
    return count;
}

async function saveAllPositions() {
    const saveStatus = document.getElementById('saveStatus');
    const saveBtn = document.getElementById('savePositionsBtn');

    saveBtn.disabled = true;
    saveBtn.textContent = 'Сохранение...';

    const promises = nodes.map(n => {
        const x = n.fx ?? n.x;
        const y = n.fy ?? n.y;
        if (Number.isFinite(x) && Number.isFinite(y)) {
            return savePositionToServer(n.id, x, y);
        }
        return Promise.resolve();
    });

    await Promise.all(promises);

    saveStatus.style.display = 'block';
    saveStatus.textContent = '✓ Позиции сохранены';
    setTimeout(() => saveStatus.style.display = 'none', 3000);

    saveBtn.disabled = false;
    saveBtn.textContent = 'Сохранить все';
}

// ============ DRAG ============

function drag(simulation) {
    function dragstarted(event, d) {
        if (!event.active) simulation.alphaTarget(0.3).restart();
        d.fx = d.x;
        d.fy = d.y;
    }

    function dragged(event, d) {
        d.fx = event.x;
        d.fy = event.y;
    }

    function dragended(event, d) {
        if (!event.active) simulation.alphaTarget(0);
        savePositionToServer(d.id, d.fx, d.fy).then(showSaveIndicator);
    }

    return d3.drag()
        .on("start", dragstarted)
        .on("drag", dragged)
        .on("end", dragended);
}

function initializeFilters() {
    const typeCounts = {};
    let onlineCount = 0, offlineCount = 0, warningCount = 0;

    allNodes.forEach(node => {
        const nodeData = node.data || node;
        const type = nodeData.device_type || 'unknown';
        typeCounts[type] = (typeCounts[type] || 0) + 1;
        filterState.allDeviceTypes.add(type);
        filterState.deviceTypes.add(type);

        if (nodeData.status === false) offlineCount++;
        else onlineCount++;

        if (nodeData.status_serv === 'critical' || nodeData.status_serv === 'warning') {
            warningCount++;
        }
    });

    document.getElementById('countOnline').textContent = onlineCount;
    document.getElementById('countOffline').textContent = offlineCount;
    document.getElementById('countWarning').textContent = warningCount;

    const container = document.getElementById('deviceTypeFilters');
    container.innerHTML = '';

    // Все устройства
    const allItem = document.createElement('div');
    allItem.className = 'filter-item active';
    allItem.dataset.type = 'all';
    allItem.innerHTML = `
        <div class="filter-checkbox"></div>
        <div class="filter-icon icon-all">📦</div>
        <span class="filter-label">Все устройства</span>
        <span class="filter-count">${allNodes.length}</span>
    `;
    allItem.addEventListener('click', () => toggleDeviceTypeFilter('all', allItem));
    container.appendChild(allItem);

    // По типам
    Object.entries(typeCounts).sort((a, b) => b[1] - a[1]).forEach(([type, count]) => {
        const config = getDeviceTypeConfig(type);
        const item = document.createElement('div');
        item.className = 'filter-item active';
        item.dataset.type = type;
        item.innerHTML = `
            <div class="filter-checkbox"></div>
            <div class="filter-icon ${config.class}">${config.icon}</div>
            <span class="filter-label">${config.label}</span>
            <span class="filter-count">${count}</span>
        `;
        item.addEventListener('click', () => toggleDeviceTypeFilter(type, item));
        container.appendChild(item);
    });

    updateFilterStatus();
}

function toggleDeviceTypeFilter(type, element) {
    if (type === 'all') {
        const allActive = filterState.deviceTypes.size === filterState.allDeviceTypes.size;
        if (allActive) {
            filterState.deviceTypes.clear();
        } else {
            filterState.allDeviceTypes.forEach(t => filterState.deviceTypes.add(t));
        }
        document.querySelectorAll('#deviceTypeFilters .filter-item').forEach(item => {
            item.classList.toggle('active', !allActive || item.dataset.type !== 'all');
        });
    } else {
        if (filterState.deviceTypes.has(type)) {
            filterState.deviceTypes.delete(type);
            element.classList.remove('active');
        } else {
            filterState.deviceTypes.add(type);
            element.classList.add('active');
        }
    }
    applyFilters();
}

function setupQuickFilters() {

            document.querySelectorAll('.quick-filter-btn').forEach(b =>
                b.classList.remove('active'));
            btn.classList.add('active');
            filterState.quickFilter = btn.dataset.filter;

            // Reset
            filterState.allDeviceTypes.forEach(t => filterState.deviceTypes.add(t));
            filterState.statusFilter = 'all';

            if (btn.dataset.filter === 'online') {
                filterState.statusFilter = 'online';
            } else if (btn.dataset.filter === 'endpoints') {
                filterState.deviceTypes.clear();
                filterState.allDeviceTypes.forEach(type => {
                    const config = getDeviceTypeConfig(type);
                    if (config.group === 'endpoints') {
                        filterState.deviceTypes.add(type);
                    }
                });
            }

            applyFilters();
        });
    });
}

function checkNodeVisibility(node) {
    const nodeData = node.data || node;
    const type = nodeData.device_type || 'unknown';

    if (!filterState.deviceTypes.has(type)) return false;

    if (filterState.statusFilter === 'online' && nodeData.status === false) return false;
    if (filterState.statusFilter === 'offline' && nodeData.status !== false) return false;
    if (filterState.statusFilter === 'warning' &&
        nodeData.status_serv !== 'critical' && nodeData.status_serv !== 'warning') return false;

    return true;
}

function applyFilters() {
    let visibleCount = 0;
    const visibleIds = new Set();

    nodeElements.each(function(d) {
        const isVisible = checkNodeVisibility(d);
        d3.select(this).style("display", isVisible ? null : "none");
        if (isVisible) {
            visibleCount++;
            visibleIds.add(d.id);
        }
    });

    linkElements.each(function(l) {
        const sourceId = typeof l.source === 'object' ? l.source.id : l.source;
        const targetId = typeof l.target === 'object' ? l.target.id : l.target;
        const isVisible = visibleIds.has(sourceId) && visibleIds.has(targetId);
        d3.select(this).style("display", isVisible ? null : "none");
    });

    updateFilterStatus(visibleCount);
}

function updateFilterStatus(visibleCount = null) {
    if (visibleCount === null) {
        visibleCount = allNodes.filter(n => checkNodeVisibility(n)).length;
    }
    document.getElementById('visibleCount').textContent = visibleCount;
    document.getElementById('totalCount').textContent = allNodes.length;
}

        item.classList.add('active');
    });
    document.querySelectorAll('#statusFilters .filter-item').forEach(item => {
        item.classList.toggle('active', item.dataset.status === 'all');
    });

    applyFilters();
}

function invertFilters() {
    const newSelection = new Set();
    filterState.allDeviceTypes.forEach(type => {
        if (!filterState.deviceTypes.has(type)) newSelection.add(type);
    });
    if (newSelection.size === 0) {
        filterState.allDeviceTypes.forEach(t => newSelection.add(t));
    }
    filterState.deviceTypes = newSelection;

    document.querySelectorAll('#deviceTypeFilters .filter-item').forEach(item => {
        if (item.dataset.type !== 'all') {
            item.classList.toggle('active', filterState.deviceTypes.has(item.dataset.type));
        }
    });

    applyFilters();
}

function searchNodes(query) {
    if (!query.trim()) return [];
    const term = query.toLowerCase();

    return allNodes.filter(node => {
        if (!checkNodeVisibility(node)) return false;
        const nodeData = node.data || node;
        const name = (nodeData.name || nodeData.hostname || '').toLowerCase();
        const ip = (nodeData.ipaddr || '').toLowerCase();
        return name.includes(term) || ip.includes(term);
    });
}

function displaySearchResults(results) {
    const container = document.getElementById('searchResults');

    if (results.length === 0) {
        container.innerHTML = '<div class="search-result-item">Ничего не найдено</div>';
        container.style.display = 'block';
        return;
    }

    container.innerHTML = results.slice(0, 15).map((node, i) => {
        const nodeData = node.data || node;
        const name = nodeData.name || nodeData.hostname || 'No name';
        const ip = nodeData.ipaddr;
        return `<div class="search-result-item" data-index="${i}">
            <div><strong>${name}</strong></div>
            <div style="font-size: 11px; opacity: 0.7;">${ip}</div>
        </div>`;
    }).join('');

    container.style.display = 'block';

    container.querySelectorAll('.search-result-item[data-index]').forEach(item => {
        item.addEventListener('click', () => {
            const node = results[parseInt(item.dataset.index)];
            centerOnNode(node, () => highlightNodeWithEffect(node));
            container.style.display = 'none';
        });
    });
}

function centerOnNode(node, callback) {
    const scale = 1.5;
    const x = -node.x * scale + width / 2;
    const y = -node.y * scale + height / 2;

    svg.transition()
        .duration(800)
        .call(zoom.transform, d3.zoomIdentity.translate(x, y).scale(scale))
        .on("end", callback);
}

function highlightNodeWithEffect(node, duration = 5000) {
    clearHighlightEffect();
    highlightedNode = node;

    nodeElements.style("opacity", n => isConnected(node, n) ? 1 : 0.2);
    linkElements.style("opacity", l => {
        const sourceId = typeof l.source === 'object' ? l.source.id : l.source;
        const targetId = typeof l.target === 'object' ? l.target.id : l.target;
        return (sourceId === node.id || targetId === node.id) ? 1 : 0.1;
    });

    const nodeEl = nodeElements.filter(d => d.id === node.id);
    nodeEl.select('.base').style("transform", "scale(1.5)");

    highlightTimeout = setTimeout(clearHighlightEffect, duration);
}

function clearHighlightEffect() {
    if (highlightTimeout) clearTimeout(highlightTimeout);
    highlightedNode = null;
    nodeElements.style("opacity", 1);
    linkElements.style("opacity", 1);
    nodeElements.select('.base').style("transform", null);
}

function clearSearch() {
    document.getElementById('nodeSearch').value = '';
    document.getElementById('clearSearch').style.display = 'none';
    document.getElementById('searchResults').style.display = 'none';
}

    document.getElementById(modalId).classList.remove('active');

}

function closePingModal() {
    document.getElementById('pingModal').classList.remove('active');
    stopPing();
}

function stopPing() {
    if (pingInterval) {
        clearInterval(pingInterval);
        pingInterval = null;
    }
    pingCount = 0;
}

async function startPing() {
    if (!currentHost) return;
    stopPing();

    const nodeData = currentHost.data || currentHost;
    const pingModal = document.getElementById('pingModal');
    const pingOutput = document.getElementById('pingOutput');

    document.getElementById('pingTitle').textContent = `Ping: ${nodeData.ipaddr}`;
    pingOutput.innerHTML = '<div class="ping-line ping-info">Starting...</div>';
    pingModal.classList.add('active');

    async function doPing() {
        if (pingCount >= maxPings) {
            stopPing();
            return;
        }

        try {
            const resp = await fetch(`/ping/${nodeData.ipaddr}/`);
            const data = await resp.json();
            pingCount++;

            const line = document.createElement('div');
            line.className = 'ping-line ' + (data.output?.includes('time=') ? 'ping-success' : 'ping-error');
            line.textContent = data.output || `Attempt ${pingCount}`;
            pingOutput.appendChild(line);
            pingOutput.scrollTop = pingOutput.scrollHeight;
        } catch (e) {
            const line = document.createElement('div');
            line.className = 'ping-line ping-error';
            line.textContent = 'Error: ' + e.message;
            pingOutput.appendChild(line);
        }
    }

    doPing();
    pingInterval = setInterval(doPing, 1000);
}

async function init() {
    console.log("=== ИНИЦИАЛИЗАЦИЯ ===");

    // Загружаем позиции
    savedPositions = await loadPositionsFromServer();
    const appliedCount = applyLoadedPositions(savedPositions);
    positionsLoaded = appliedCount > 0;

    // Создаём симуляцию
    simulation = d3.forceSimulation(nodes)
        .force("link", d3.forceLink(links).id(d => d.id).distance(120).strength(0.5))
        .force("charge", d3.forceManyBody().strength(-400))
        .force("center", d3.forceCenter(width / 2, height / 2))
        .force("collision", d3.forceCollide().radius(35));

    // Drag
    nodeElements.call(drag(simulation));

    // Tick
    simulation.on("tick", () => {
        linkElements
            .attr("x1", d => d.source.x)
            .attr("y1", d => d.source.y)
            .attr("x2", d => d.target.x)
            .attr("y2", d => d.target.y);

        nodeElements.attr("transform", d => `translate(${d.x}, ${d.y})`);
    });

    if (positionsLoaded) {
        simulation.alpha(0.1);
    }

    // Фильтры
    initializeFilters();
    setupStatusFilters();
    setupQuickFilters();

    // Кнопки
    document.getElementById('resetPositionsBtn').addEventListener('click', resetAllPositions);
    document.getElementById('savePositionsBtn').addEventListener('click', saveAllPositions);
    document.getElementById('autoArrangeBtn').addEventListener('click', autoArrange);
    document.getElementById('filterToggle').addEventListener('click', () => {
        document.getElementById('filterContent').classList.toggle('collapsed');
        document.getElementById('filterToggle').classList.toggle('collapsed');
    });
    document.getElementById('resetFilters').addEventListener('click', resetFilters);
    document.getElementById('invertFilters').addEventListener('click', invertFilters);

    // Поиск
    document.getElementById('searchBtn').addEventListener('click', () => {
        const q = document.getElementById('nodeSearch').value;
        if (q) {
            displaySearchResults(searchNodes(q));
            document.getElementById('clearSearch').style.display = 'block';
        }
    });

    document.getElementById('nodeSearch').addEventListener('input', (e) => {
        if (e.target.value) {
            displaySearchResults(searchNodes(e.target.value));
            document.getElementById('clearSearch').style.display = 'block';
        } else {
            clearSearch();
        }
    });

    document.getElementById('nodeSearch').addEventListener('keypress', (e) => {
        if (e.key === 'Enter') {
            const results = searchNodes(e.target.value);
            if (results.length > 0) {
                centerOnNode(results[0], () => highlightNodeWithEffect(results[0]));
                document.getElementById('searchResults').style.display = 'none';
            }
        }
    });

    document.getElementById('clearSearch').addEventListener('click', () => {
        clearSearch();
        clearHighlightEffect();
    });

    // Resize
    window.addEventListener('resize', () => {
        width = window.innerWidth;
        height = window.innerHeight;
        svg.attr("width", width).attr("height", height);
        simulation.force("center", d3.forceCenter(width / 2, height / 2));
        simulation.alpha(0.1).restart();
    });

    console.log("=== ИНИЦИАЛИЗАЦИЯ ЗАВЕРШЕНА ===");
    console.log("Узлов отображено:", nodeElements.size());
}

// Запуск
init();
